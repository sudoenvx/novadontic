model User {
  id                   BigInt                 @id @default(autoincrement()) @db.BigInt
  tenantId             BigInt                 @map("tenant_id") @db.BigInt
  fullName             String                 @map("full_name") @db.VarChar(150)
  email                String                 @db.VarChar(150)
  passwordHash         String                 @map("password_hash")
  phone                String?                @db.VarChar(30)
  avatarUrl            String?                @map("avatar_url")
  isActive             Boolean                @default(true) @map("is_active")
  failedLoginCount     Int                    @default(0) @map("failed_login_count") @db.SmallInt
  lockedUntil          DateTime?              @map("locked_until") @db.Timestamptz(6)
  mfaEnabled           Boolean                @default(false) @map("mfa_enabled")
  mfaSecretEncrypted   Bytes?                 @map("mfa_secret_encrypted") @db.ByteA
  lastLoginAt          DateTime?              @map("last_login_at") @db.Timestamptz(6)
  createdAt            DateTime               @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt            DateTime               @default(now()) @updatedAt @map("updated_at") @db.Timestamptz(6)
  tenant               Tenant                 @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  roles                UserRole[]
  staffSessions        AuthSession[]          @relation("StaffSessions")
  impersonationTargets ImpersonationSession[] @relation("StaffImpersonationTarget")
  auditLogs            AuditLog[]             @relation("StaffAuditActor")

  @@unique([tenantId, id])
  @@unique([tenantId, email])
  @@index([tenantId, isActive])
  @@map("users")
}

model Role {
  id          BigInt           @id @default(autoincrement()) @db.BigInt
  tenantId    BigInt           @map("tenant_id") @db.BigInt
  code        String           @db.VarChar(40)
  name        String           @db.VarChar(100)
  isSystem    Boolean          @default(false) @map("is_system")
  tenant      Tenant           @relation(fields: [tenantId], references: [id], onDelete: Cascade)
  users       UserRole[]
  permissions RolePermission[]

  @@unique([tenantId, id])
  @@unique([tenantId, code])
  @@map("roles")
}

model Permission {

  module      String           @db.VarChar(40)

  @@map("permissions")
}

model RolePermission {

  permission   Permission @relation(fields: [permissionId], references: [id], onDelete: Cascade)

  @@id([roleId, permissionId])
  @@index([tenantId])
  @@map("role_permissions")
}

model UserRole {
